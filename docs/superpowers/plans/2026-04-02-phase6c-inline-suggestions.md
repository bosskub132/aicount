# Phase 6C: Inline AI Suggestions — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add inline AI suggestions (COA mapping, WHT rate, duplicate detection, smart defaults) to the extraction workflow with below-field pills, two-pass duplicate detection, and full suggestion lifecycle tracking.

**Architecture:** Dedicated suggestion service layer (`src/lib/services/suggestions/`) with coordinator + provider pattern mirroring the extraction pipeline. Cheap suggestions (vendor history, graduated rules) pre-computed eagerly during Inngest extraction. AI-powered suggestions lazy-loaded on page open. All interactions tracked in local state and batch-submitted with form save.

**Tech Stack:** Drizzle ORM (schema + queries), Inngest (pipeline step), React Query (data fetching), Zustand/useState (local suggestion state), Claude Haiku (AI COA fallback), SHA-256 (file hash dedup)

```typescript
export async function findContentMatches(
  tenantId: string,
  documentId: string,
  issuerTaxId: string,
  grandTotal: number,
  documentDate: string
): Promise<{ id: string; issuerName: string | null; documentNumber: string | null; documentDate: string | null; grandTotal: string | null; status: string }[]> {
  const amountLow = grandTotal * 0.95;
  const amountHigh = grandTotal * 1.05;
  const dateObj = new Date(documentDate);
  const dateLow = new Date(dateObj.getTime() - 30 * 24 * 60 * 60 * 1000);
  const dateHigh = new Date(dateObj.getTime() + 30 * 24 * 60 * 60 * 1000);

  return db
    .select({
      id: documents.id,
      issuerName: documents.issuerName,
      documentNumber: documents.documentNumber,
      documentDate: documents.documentDate,
      grandTotal: documents.grandTotal,
      status: documents.status,
    })
    .from(documents)
    .where(
      and(
        eq(documents.tenantId, tenantId),
        ne(documents.id, documentId),
        eq(documents.issuerTaxId, issuerTaxId),
        gte(documents.grandTotal, String(amountLow)),
        lte(documents.grandTotal, String(amountHigh)),
        gte(documents.documentDate, dateLow.toISOString().slice(0, 10)),
        lte(documents.documentDate, dateHigh.toISOString().slice(0, 10))
      )
    )
    .limit(10);
}
```

```typescript
// src/lib/services/suggestions/providers/coa-suggester.ts

import { and, eq, desc, sql, ne } from "drizzle-orm";
import { db } from "@/lib/db";
import {
  journalLines,
  journalEntries,
  documents,
  chartOfAccounts,
} from "@/lib/db/schema";
import type { SuggestionResult } from "../types";

export async function fromHistory(
  documentId: string,
  tenantId: string
): Promise<SuggestionResult[]> {
  // Get the issuer tax ID from the current document
  const [doc] = await db
    .select({ issuerTaxId: documents.issuerTaxId })
    .from(documents)
    .where(eq(documents.id, documentId))
    .limit(1);

  if (!doc?.issuerTaxId) return [];

  // Find most frequent GL account used for this vendor in posted journal entries
  const rows = await db
    .select({
      accountCode: journalLines.accountCode,
      usageCount: sql<number>`count(*)`.as("usage_count"),
    })
    .from(journalLines)
    .innerJoin(
      journalEntries,
      eq(journalLines.journalEntryId, journalEntries.id)
    )
    .innerJoin(documents, eq(journalLines.documentId, documents.id))
    .where(
      and(
        eq(journalEntries.tenantId, tenantId),
        eq(journalEntries.status, "posted"),
        eq(documents.issuerTaxId, doc.issuerTaxId),
        ne(documents.id, documentId),
        sql`${journalLines.accountCode} IS NOT NULL`
      )
    )
    .groupBy(journalLines.accountCode)
    .orderBy(desc(sql`count(*)`))
    .limit(1);

  if (rows.length === 0) return [];

  const topAccount = rows[0];

  // Get the account name for display
  const [account] = await db
    .select({ name: chartOfAccounts.name })
    .from(chartOfAccounts)
    .where(
      and(
        eq(chartOfAccounts.tenantId, tenantId),
        eq(chartOfAccounts.code, topAccount.accountCode!)
      )
    )
    .limit(1);

  // Calculate total docs for this vendor to get confidence
  const [totalRow] = await db
    .select({ total: sql<number>`count(distinct ${documents.id})` })
    .from(journalLines)
    .innerJoin(
      journalEntries,
      eq(journalLines.journalEntryId, journalEntries.id)
    )
    .innerJoin(documents, eq(journalLines.documentId, documents.id))
    .where(
      and(
        eq(journalEntries.tenantId, tenantId),
        eq(journalEntries.status, "posted"),
        eq(documents.issuerTaxId, doc.issuerTaxId),
        ne(documents.id, documentId)
      )
    );

  const total = Number(totalRow?.total ?? 0);
  const frequency = Number(topAccount.usageCount);
  const confidence = total > 0 ? Math.min(0.99, frequency / total) : 0.5;

  return [
    {
      feature: "coa_mapping",
      fieldName: "glAccountCode",
      suggestedValue: topAccount.accountCode!,
      confidence,
      source: "vendor_history",
      sourceContext: {
        accountName: account?.name ?? topAccount.accountCode,
        displayLabel: `${topAccount.accountCode} ${account?.name ?? ""}`.trim(),
        frequency,
        totalDocs: total,
        issuerTaxId: doc.issuerTaxId,
      },
    },
  ];
}

/**
 * AI fallback: When no vendor history exists, use Claude Haiku to suggest
 * a GL account based on line item descriptions + tenant's chart of accounts.
 * Called lazily (only on page open).
 */
export async function full(
  documentId: string,
  tenantId: string
): Promise<SuggestionResult[]> {
  // First try history
  const historyResults = await fromHistory(documentId, tenantId);
  if (historyResults.length > 0) return historyResults;

  // No history — use AI
  const [doc] = await db
    .select({ ocrRaw: documents.ocrRaw })
    .from(documents)
    .where(eq(documents.id, documentId))
    .limit(1);

  if (!doc?.ocrRaw) return [];

  const ocrRaw = doc.ocrRaw as Record<string, unknown>;
  const lineItems = (ocrRaw.line_items ?? []) as { description?: string }[];
  if (lineItems.length === 0) return [];

  // Get tenant's chart of accounts
  const accounts = await db
    .select({ code: chartOfAccounts.code, name: chartOfAccounts.name })
    .from(chartOfAccounts)
    .where(eq(chartOfAccounts.tenantId, tenantId))
    .limit(100);

  if (accounts.length === 0) return [];

  const descriptions = lineItems
    .map((li) => li.description)
    .filter(Boolean)
    .join(", ");

  const accountList = accounts
    .map((a) => `${a.code} ${a.name}`)
    .join("\n");

  try {
    const { default: Anthropic } = await import("@anthropic-ai/sdk");
    const client = new Anthropic();

    const startTime = Date.now();
    const response = await client.messages.create({
      model: "claude-haiku-4-5-20251001",
      max_tokens: 200,
      messages: [
        {
          role: "user",
          content: `Given these line item descriptions from a Thai accounting document:\n${descriptions}\n\nWhich GL account is the best match? Pick exactly one from this chart of accounts:\n${accountList}\n\nRespond with ONLY the account code (e.g., "5310"). No explanation.`,
        },
      ],
    });

    const text =
      response.content[0].type === "text" ? response.content[0].text.trim() : "";
    const matchedAccount = accounts.find(
      (a) => a.code === text || text.startsWith(a.code)
    );

    // Log usage
    const { logAiUsage } = await import(
      "@/lib/services/extraction/usage-logger"
    );
    await logAiUsage({
      tenantId,
      documentId,
      provider: "anthropic",
      model: "claude-haiku-4-5-20251001",
      feature: "suggestion",
      tier: null,
      inputTokens: response.usage.input_tokens,
      outputTokens: response.usage.output_tokens,
      costUsd:
        response.usage.input_tokens * 0.0000008 +
        response.usage.output_tokens * 0.000004,
      metadata: { type: "coa_suggestion", descriptions, suggestedCode: text },
    });

    if (!matchedAccount) return [];

    return [
      {
        feature: "coa_mapping",
        fieldName: "glAccountCode",
        suggestedValue: matchedAccount.code,
        confidence: 0.65,
        source: "ai_model",
        sourceContext: {
          displayLabel: `${matchedAccount.code} ${matchedAccount.name}`,
          accountName: matchedAccount.name,
          model: "claude-haiku-4-5-20251001",
          descriptions,
        },
      },
    ];
  } catch {
    // AI failure is non-blocking
    return [];
  }
}
```

```typescript
// src/lib/services/suggestions/providers/wht-suggester.ts

import { and, eq, desc, sql, ne, isNotNull } from "drizzle-orm";
import { db } from "@/lib/db";
import { documents, vendors } from "@/lib/db/schema";
import type { SuggestionResult } from "../types";

export async function fromHistory(
  documentId: string,
  tenantId: string
): Promise<SuggestionResult[]> {
  const [doc] = await db
    .select({ issuerTaxId: documents.issuerTaxId })
    .from(documents)
    .where(eq(documents.id, documentId))
    .limit(1);

  if (!doc?.issuerTaxId) return [];

  // Find most frequent WHT rate for this vendor
  const rows = await db
    .select({
      whtRate: documents.whtRate,
      whtIncomeType: documents.whtIncomeType,
      usageCount: sql<number>`count(*)`.as("usage_count"),
    })
    .from(documents)
    .where(
      and(
        eq(documents.tenantId, tenantId),
        eq(documents.issuerTaxId, doc.issuerTaxId),
        ne(documents.id, documentId),
        isNotNull(documents.whtRate)
      )
    )
    .groupBy(documents.whtRate, documents.whtIncomeType)
    .orderBy(desc(sql`count(*)`))
    .limit(1);

  if (rows.length === 0) return [];

  const top = rows[0];

  const [totalRow] = await db
    .select({ total: sql<number>`count(*)` })
    .from(documents)
    .where(
      and(
        eq(documents.tenantId, tenantId),
        eq(documents.issuerTaxId, doc.issuerTaxId),
        ne(documents.id, documentId),
        isNotNull(documents.whtRate)
      )
    );

  const total = Number(totalRow?.total ?? 0);
  const frequency = Number(top.usageCount);
  const confidence = total > 0 ? Math.min(0.99, frequency / total) : 0.5;

  const results: SuggestionResult[] = [];

  if (top.whtRate) {
    results.push({
      feature: "wht_rate",
      fieldName: "whtRate",
      suggestedValue: top.whtRate,
      confidence,
      source: "vendor_history",
      sourceContext: {
        whtIncomeType: top.whtIncomeType,
        displayLabel: `${top.whtRate}% ${top.whtIncomeType ?? ""}`.trim(),
        frequency,
        totalDocs: total,
      },
    });
  }

  if (top.whtIncomeType) {
    results.push({
      feature: "wht_rate",
      fieldName: "whtIncomeType",
      suggestedValue: top.whtIncomeType,
      confidence,
      source: "vendor_history",
      sourceContext: {
        whtRate: top.whtRate,
        frequency,
        totalDocs: total,
      },
    });
  }

  return results;
}

export async function fromVendorType(
  documentId: string,
  tenantId: string
): Promise<SuggestionResult[]> {
  const [doc] = await db
    .select({ issuerTaxId: documents.issuerTaxId })
    .from(documents)
    .where(eq(documents.id, documentId))
    .limit(1);

  if (!doc?.issuerTaxId) return [];

  const [vendor] = await db
    .select({ vendorType: vendors.vendorType })
    .from(vendors)
    .where(
      and(eq(vendors.tenantId, tenantId), eq(vendors.taxId, doc.issuerTaxId))
    )
    .limit(1);

  if (!vendor) return [];

  // Default WHT rates by vendor type (Thai tax rules)

    vendor.vendorType === "individual" ? "40(2) Services" : "40(2) Services";

  return [
    {
      feature: "wht_rate",
      fieldName: "whtRate",
      suggestedValue: defaultRate,
      confidence: 0.6,
      source: "frequency",
      sourceContext: {
        vendorType: vendor.vendorType,
        displayLabel: `${defaultRate}% ${defaultType}`,
        reason: `Default for ${vendor.vendorType} vendors`,
      },
    },
  ];
}
```

```typescript
export function computeFileHash(fileBuffer: Buffer): string {
  return createHash("sha256").update(fileBuffer).digest("hex");
}

export async function checkFileHash(
  tenantId: string,
  documentId: string,
  fileHash: string
): Promise<void> {
  const matches = await findByFileHash(tenantId, fileHash, documentId);

  for (const match of matches) {
    await insertDuplicateCandidate(
      tenantId,
      documentId,
      match.id,
      "file_hash",
      1.0,
      { reason: "Identical file content (SHA-256 match)" }
    );
  }
}

export async function contentMatch(
  documentId: string,
  tenantId: string
): Promise<SuggestionResult[]> {
  const [doc] = await db
    .select({
      issuerTaxId: documents.issuerTaxId,
      grandTotal: documents.grandTotal,
      documentDate: documents.documentDate,
    })
    .from(documents)
    .where(eq(documents.id, documentId))
    .limit(1);

  if (!doc?.issuerTaxId || !doc.grandTotal || !doc.documentDate) return [];

  const matches = await findContentMatches(
    tenantId,
    documentId,
    doc.issuerTaxId,
    Number(doc.grandTotal),
    doc.documentDate
  );

  for (const match of matches) {
    const fieldsMatched: string[] = [];
    if (match.issuerName) fieldsMatched.push("issuer");
    if (match.grandTotal && Math.abs(Number(match.grandTotal) - Number(doc.grandTotal)) < 0.01) {
      fieldsMatched.push("exact_amount");
    } else {
      fieldsMatched.push("similar_amount");
    }
    if (match.documentDate === doc.documentDate) fieldsMatched.push("same_date");

    const score = Math.min(0.99, 0.5 + fieldsMatched.length * 0.15);

    await insertDuplicateCandidate(
      tenantId,
      documentId,
      match.id,
      "content_match",
      score,
      { fieldsMatched, matchedDoc: match }
    );
  }

  return [];
}
```

```typescript
import { and, eq, desc, sql, ne, isNotNull } from "drizzle-orm";
import { db } from "@/lib/db";
import { documents, journalEntries } from "@/lib/db/schema";

interface FrequencyRow {
  value: string;
  usageCount: number;
}

async function getMostFrequentValue(
  tenantId: string,
  issuerTaxId: string,
  documentId: string,
  column: typeof documents.direction
): Promise<FrequencyRow | null> {
  const rows = await db
    .select({
      value: column,
      usageCount: sql<number>`count(*)`.as("usage_count"),
    })
    .from(documents)
    .where(
      and(
        eq(documents.tenantId, tenantId),
        eq(documents.issuerTaxId, issuerTaxId),
        ne(documents.id, documentId),
        isNotNull(column)
      )
    )
    .groupBy(column)
    .orderBy(desc(sql`count(*)`))
    .limit(1);

  if (rows.length === 0 || !rows[0].value) return null;
  return { value: rows[0].value as string, usageCount: Number(rows[0].usageCount) };
}

export async function fromHistory(
  documentId: string,
  tenantId: string
): Promise<SuggestionResult[]> {
  const [doc] = await db
    .select({ issuerTaxId: documents.issuerTaxId })
    .from(documents)
    .where(eq(documents.id, documentId))
    .limit(1);

  if (!doc?.issuerTaxId) return [];

  // Count total docs for this vendor
  const [totalRow] = await db
    .select({ total: sql<number>`count(*)` })
    .from(documents)
    .where(
      and(
        eq(documents.tenantId, tenantId),
        eq(documents.issuerTaxId, doc.issuerTaxId),
        ne(documents.id, documentId)
      )
    );

  const total = Number(totalRow?.total ?? 0);
  if (total === 0) return [];

  const results: SuggestionResult[] = [];

  // Suggest document type based on vendor history
  const docType = await getMostFrequentValue(
    tenantId,
    doc.issuerTaxId,
    documentId,
    documents.docType
  );

  if (docType) {
    const confidence = Math.min(0.99, docType.usageCount / total);
    if (confidence >= 0.6) {
      results.push({
        feature: "smart_default",
        fieldName: "docType",
        suggestedValue: docType.value,
        confidence,
        source: "frequency",
        sourceContext: {
          displayLabel: docType.value.replace(/_/g, " "),
          frequency: docType.usageCount,
          totalDocs: total,
        },
      });
    }
  }

  // Suggest direction based on vendor history
  const direction = await getMostFrequentValue(
    tenantId,
    doc.issuerTaxId,
    documentId,
    documents.direction
  );

  if (direction) {
    const confidence = Math.min(0.99, direction.usageCount / total);
    if (confidence >= 0.6) {
      results.push({
        feature: "smart_default",
        fieldName: "direction",
        suggestedValue: direction.value,
        confidence,
        source: "frequency",
        sourceContext: {
          displayLabel: direction.value === "REVENUE" ? "Revenue (AR)" : "Expense (AP)",
          frequency: direction.usageCount,
          totalDocs: total,
        },
      });
    }
  }

  return results;
}
```

```typescript
import * as coaSuggester from "./providers/coa-suggester";
import * as whtSuggester from "./providers/wht-suggester";
import * as duplicateChecker from "./providers/duplicate-checker";
import * as defaultsSuggester from "./providers/defaults-suggester";
import {
  insertSuggestions,
  getPendingSuggestions,
  hasAnySuggestions,
} from "@/lib/db/queries/suggestions";

import type { Suggestion, SuggestionResult, DuplicateCandidate } from "./types";

type Mode = "eager" | "lazy";

export async function getSuggestions(
  documentId: string,
  tenantId: string,
  mode: Mode
): Promise<SuggestionResult[]> {
  const providers =
    mode === "eager"
      ? [
          () => coaSuggester.fromHistory(documentId, tenantId),
          () => whtSuggester.fromHistory(documentId, tenantId),
          () => defaultsSuggester.fromHistory(documentId, tenantId),
        ]
      : [
          () => coaSuggester.full(documentId, tenantId),
          () => whtSuggester.fromVendorType(documentId, tenantId),
          () => duplicateChecker.contentMatch(documentId, tenantId),
        ];

  const settled = await Promise.allSettled(providers.map((p) => p()));

  const allResults: SuggestionResult[] = [];
  for (const result of settled) {
    if (result.status === "fulfilled") {
      allResults.push(...result.value);
    }
  }

  // Deduplicate by fieldName — keep highest confidence
  const byField = new Map<string, SuggestionResult>();
  for (const r of allResults) {
    const existing = byField.get(r.fieldName);
    if (!existing || r.confidence > existing.confidence) {
      byField.set(r.fieldName, r);
    }
  }

  const deduplicated = Array.from(byField.values());

  if (deduplicated.length > 0) {
    await insertSuggestions(tenantId, documentId, deduplicated);
  }

  return deduplicated;
}

export async function fetchAllForDocument(
  documentId: string,
  tenantId: string
): Promise<{ suggestions: Suggestion[]; duplicates: DuplicateCandidate[] }> {
  // Run lazy providers if not already run
  const alreadyHasSuggestions = await hasAnySuggestions(documentId);
  if (!alreadyHasSuggestions) {
    // First time opening — run eager results were already saved, now run lazy
    await getSuggestions(documentId, tenantId, "lazy");
  }

  const [suggestions, duplicates] = await Promise.all([
    getPendingSuggestions(documentId),
    getPendingDuplicates(documentId),
  ]);

  return { suggestions, duplicates };
}
```

```typescript
// src/lib/services/suggestions/learner.ts

import { and, eq } from "drizzle-orm";
import { db } from "@/lib/db";
import { aiSuggestions, aiExtractionRules } from "@/lib/db/schema";
import type { SuggestionOutcome } from "./types";

export async function feedLearner(
  outcomes: SuggestionOutcome[]
): Promise<void> {
  for (const outcome of outcomes) {
    const [suggestion] = await db
      .select()
      .from(aiSuggestions)
      .where(eq(aiSuggestions.id, outcome.id))
      .limit(1);

    if (!suggestion) continue;

    if (outcome.status === "dismissed") {
      // Decrease confidence for rules matching this pattern
      await degradeMatchingRules(
        suggestion.tenantId,
        suggestion.fieldName,
        suggestion.suggestedValue
      );
    }

    if (outcome.status === "edited" && outcome.finalValue) {
      // User accepted but changed value — the correction signal
      // This feeds into the existing Phase 6A rule learner pattern
      await degradeMatchingRules(
        suggestion.tenantId,
        suggestion.fieldName,
        suggestion.suggestedValue
      );
    }
  }
}

async function degradeMatchingRules(
  tenantId: string,
  fieldName: string,
  suggestedValue: string
): Promise<void> {
  const rules = await db
    .select()
    .from(aiExtractionRules)
    .where(
      and(
        eq(aiExtractionRules.tenantId, tenantId),
        eq(aiExtractionRules.fieldName, fieldName)
      )
    );

  for (const rule of rules) {
    if (rule.deterministicValue === suggestedValue) {
      const newConfidence = Math.max(0.1, Number(rule.confidence) - 0.2);
      const isGraduated = newConfidence >= 0.5 ? rule.isGraduated : false;

      await db
        .update(aiExtractionRules)
        .set({
          confidence: String(newConfidence),
          isGraduated,
          updatedAt: new Date(),
        })
        .where(eq(aiExtractionRules.id, rule.id));
    }
  }
}
```

```typescript
// src/lib/hooks/use-suggestions.ts

"use client";

import { useQuery } from "@tanstack/react-query";
import { useCallback, useState } from "react";
import { getWorkspaceTenantId } from "@/components/workspace-selector";
import type {
  Suggestion,
  DuplicateCandidate,
  SuggestionOutcome,
  DuplicateOutcome,
} from "@/lib/services/suggestions/types";

const isValidTenant = (id: string) =>
  !!id && id !== "00000000-0000-0000-0000-000000000000";

interface SuggestionState {
  suggestions: Suggestion[];
  duplicates: DuplicateCandidate[];
  outcomes: Map<string, SuggestionOutcome>;
  duplicateOutcomes: Map<string, DuplicateOutcome>;
}

export function useSuggestions(documentId: string | undefined) {
  const tenantId = getWorkspaceTenantId();

  const [outcomes, setOutcomes] = useState<Map<string, SuggestionOutcome>>(
    new Map()
  );
  const [duplicateOutcomes, setDuplicateOutcomes] = useState<
    Map<string, DuplicateOutcome>
  >(new Map());

  const query = useQuery({
    queryKey: ["suggestions", documentId, tenantId],
    queryFn: async () => {
      const res = await fetch(`/api/documents/${documentId}/suggestions`, {
        headers: { "x-tenant-id": tenantId },
      });
      const json = await res.json();
      if (!json.success) throw new Error(json.error);
      return json.data as {
        suggestions: Suggestion[];
        duplicates: DuplicateCandidate[];
      };
    },
    enabled: isValidTenant(tenantId) && !!documentId,
  });

  // Filter out already-resolved suggestions from local state
  const pendingSuggestions = (query.data?.suggestions ?? []).filter(
    (s) => !outcomes.has(s.id)
  );

  const pendingDuplicates = (query.data?.duplicates ?? []).filter(
    (d) => !duplicateOutcomes.has(d.id)
  );

  const acceptSuggestion = useCallback((id: string) => {
    setOutcomes((prev) => {
      const next = new Map(prev);
      next.set(id, { id, status: "accepted" });
      return next;
    });
  }, []);

  const dismissSuggestion = useCallback((id: string) => {
    setOutcomes((prev) => {
      const next = new Map(prev);
      next.set(id, { id, status: "dismissed" });
      return next;
    });
  }, []);

  const editSuggestion = useCallback((id: string, finalValue: string) => {
    setOutcomes((prev) => {
      const next = new Map(prev);
      next.set(id, { id, status: "edited", finalValue });
      return next;
    });
  }, []);

  const acceptAll = useCallback(() => {
    const allSuggestions = query.data?.suggestions ?? [];
    setOutcomes((prev) => {
      const next = new Map(prev);
      for (const s of allSuggestions) {
        if (!next.has(s.id)) {
          next.set(s.id, { id: s.id, status: "accepted" });
        }
      }
      return next;
    });
  }, [query.data?.suggestions]);

  const clearAll = useCallback(() => {
    const allSuggestions = query.data?.suggestions ?? [];
    setOutcomes((prev) => {
      const next = new Map(prev);
      for (const s of allSuggestions) {
        if (!next.has(s.id)) {
          next.set(s.id, { id: s.id, status: "dismissed" });
        }
      }
      return next;
    });
  }, [query.data?.suggestions]);

  const dismissDuplicate = useCallback((id: string) => {
    setDuplicateOutcomes((prev) => {
      const next = new Map(prev);
      next.set(id, { id, status: "dismissed" });
      return next;
    });
  }, []);

  // Get batch outcomes for form submission
  const getBatchOutcomes = useCallback(() => {
    return {
      suggestionOutcomes: Array.from(outcomes.values()),
      duplicateOutcomes: Array.from(duplicateOutcomes.values()),
    };
  }, [outcomes, duplicateOutcomes]);

  return {
    suggestions: pendingSuggestions,
    duplicates: pendingDuplicates,
    allSuggestions: query.data?.suggestions ?? [],
    isLoading: query.isLoading,
    acceptSuggestion,
    dismissSuggestion,
    editSuggestion,
    acceptAll,
    clearAll,
    dismissDuplicate,
    getBatchOutcomes,
  };
}
```
